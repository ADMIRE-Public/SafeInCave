 83def project(tensor_ufl: ufl.core.expr.Expr, V: do.fem.FunctionSpace) -> do.fem.Function:
 84	"""
 85	Interpolate a UFL expression into a DOLFINx function space.
 86
 87	Parameters
 88	----------
 89	tensor_ufl : ufl.core.expr.Expr
 90	    UFL expression to be interpolated (e.g., rank-2 tensor field).
 91	V : dolfinx.fem.FunctionSpace
 92	    Target function space. Its element must be compatible with the
 93	    UFL expression shape.
 94
 95	Returns
 96	-------
 97	dolfinx.fem.Function
 98	    Function in `V` obtained by interpolation of `tensor_ufl`.
 99
100	Notes
101	-----
102	This uses `dolfinx.fem.Expression` with `V.element.interpolation_points()`
103	and then calls `Function.interpolate(...)`. For non-interpolatory elements,
104	prefer projection/solve-based approaches.
105	"""
106	tensor_expr = do.fem.Expression(tensor_ufl, V.element.interpolation_points())
107	tensor = do.fem.Function(V)
108	tensor.interpolate(tensor_expr)
109	return tensor

111def epsilon(u: UFLVector3) -> UFLTensor3x3:
112	"""
113	Small-strain tensor :math:`\\varepsilon(u) = \\mathrm{sym}(\\nabla u)`.
114
115	Parameters
116	----------
117	u : UFLVector3
118	    Displacement field as a UFL vector expression with shape (3,).
119
120	Returns
121	-------
122	UFLTensor3x3
123	    Symmetric gradient of `u`, a (3, 3) UFL tensor.
124
125	Raises
126	------
127	AssertionError
128	    If `u` is not rank-1 with shape (3,).
129
130	Notes
131	-----
132	UFLVector3 = ufl.core.expr.Expr
133	"""
134	assert ufl.rank(u) == 1 and u.ufl_shape == (3,)
135	grad_u = ufl.sym(ufl.grad(u))
136	return grad_u

138def dotdot_ufl(C: UFLMatrix6x6, eps: UFLTensor3x3) -> UFLTensor3x3:
139	"""
140	Compute stress :math:`\\sigma = C : \\varepsilon` using Voigt mapping in UFL.
141
142	Parameters
143	----------
144	C : UFLMatrix6x6
145	    Fourth-order stiffness tensor stored in Voigt form as a (6, 6) UFL object.
146	    Assumes **tensorial Voigt** convention (no engineering shear factors).
147	eps : UFLTensor3x3
148	    Small-strain tensor as a (3, 3) UFL expression.
149
150	Returns
151	-------
152	UFLTensor3x3
153	    Cauchy stress tensor as a (3, 3) UFL expression.
154
155	Raises
156	------
157	AssertionError
158	    If shapes are not `(6, 6)` for `C` or `(3, 3)` for `eps`.
159
160	Notes
161	-----
162	Internally computes `voigt2tensor(ufl.dot(C, tensor2voigt(eps)))`.
163	UFLMatrix6x6 = ufl.core.expr.Expr
164	UFLTensor3x3 = ufl.core.expr.Expr
165	"""
166	assert ufl.rank(C) == 2 and C.ufl_shape == (6, 6)
167	assert ufl.rank(eps) == 2 and eps.ufl_shape == (3, 3)
168	tensor = voigt2tensor(ufl.dot(C, tensor2voigt(eps)))
169	return tensor

171def tensor2voigt(e: UFLTensor3x3) -> UFLVector6:
172	"""
173    Map a 3×3 symmetric tensor to Voigt vector (tensorial Voigt).
174
175    Parameters
176    ----------
177    e : UFLTensor3x3
178        Tensor expression with shape (3, 3). Symmetry is assumed but not enforced.
179
180    Returns
181    -------
182    UFLVector6
183        Voigt vector `[e_xx, e_yy, e_zz, e_xy, e_xz, e_yz]`.
184
185    Raises
186    ------
187    AssertionError
188        If `e` is not rank-2 with shape (3, 3).
189
190    Notes
191    -----
192    multiply the shear components by 0.5 when mapping back to tensor form.
193    UFLTensor3x3 = ufl.core.expr.Expr
194    """
195	assert ufl.rank(e) == 2 and e.ufl_shape == (3, 3)
196	e_voigt = ufl.as_vector([e[0,0], e[1,1], e[2,2], e[0,1], e[0,2], e[1,2]])
197	return e_voigt

199def voigt2tensor(s: UFLVector6) -> UFLTensor3x3:
200	"""
201    Map a Voigt vector (length 6) to a symmetric 3×3 tensor (tensorial Voigt).
202
203    Parameters
204    ----------
205    s : UFLVector6
206        Voigt vector `[xx, yy, zz, xy, xz, yz]` (no engineering shear factors).
207
208    Returns
209    -------
210    UFLTensor3x3
211        Symmetric (3, 3) tensor reconstructed from `s`.
212
213    Raises
214    ------
215    AssertionError
216        If the UFL shape of `s` is not `(6,)`.
217
218    Notes
219    -----
220    UFLVector6 = ufl.core.expr.Expr
221    UFLTensor3x3 = ufl.core.expr.Expr
222    """
223	assert s.ufl_shape == (6,)
224	s_tensor = ufl.as_matrix([[s[0], s[3], s[4]],
225							  [s[3], s[1], s[5]],
226							  [s[4], s[5], s[2]]])
227	return s_tensor

251def dotdot_torch(C_voigt: to.Tensor, eps_tensor: to.Tensor) -> to.Tensor:
252	"""
253    σ = C : ε using Voigt notation (tensorial Voigt, no engineering factors).
254
255    Parameters
256    ----------
257    C_voigt : (N, 6, 6) torch.Tensor (float64)
258        Stiffness matrix per element in tensorial Voigt order [xx, yy, zz, xy, xz, yz].
259    eps_tensor : (N, 3, 3) torch.Tensor (float64)
260        Small-strain tensor per element.
261
262    Returns
263    -------
264    (N, 3, 3) torch.Tensor (float64)
265        Cauchy stress tensor per element.
266    """
267	n_elems = C_voigt.shape[0]
268	eps_voigt = to.zeros((n_elems, 6), dtype=to.float64)
269	eps_voigt[:,0] = eps_tensor[:,0,0]
270	eps_voigt[:,1] = eps_tensor[:,1,1]
271	eps_voigt[:,2] = eps_tensor[:,2,2]
272	eps_voigt[:,3] = eps_tensor[:,0,1]
273	eps_voigt[:,4] = eps_tensor[:,0,2]
274	eps_voigt[:,5] = eps_tensor[:,1,2]
275	stress_voigt = to.bmm(C_voigt, eps_voigt.unsqueeze(2)).squeeze(2)
276	stress_torch = to.zeros_like(eps_tensor, dtype=to.float64)
277	stress_torch[:,0,0] = stress_voigt[:,0]
278	stress_torch[:,1,1] = stress_voigt[:,1]
279	stress_torch[:,2,2] = stress_voigt[:,2]
280	stress_torch[:,0,1] = stress_torch[:,1,0] = stress_voigt[:,3]
281	stress_torch[:,0,2] = stress_torch[:,2,0] = stress_voigt[:,4]
282	stress_torch[:,1,2] = stress_torch[:,2,1] = stress_voigt[:,5]
283	return stress_torch

285def create_field_nodes(grid: GridHandlerGMSH, fun: Fn) -> to.Tensor:
286	"""
287    Sample a N-dimensional field at mesh nodes using a Python callable.
288
289    Parameters
290    ----------
291    grid : GridHandlerGMSH
292        Grid handler with attributes `mesh` and `n_nodes`. Assumes
293        `grid.mesh.geometry.x` provides an array of node coordinates of shape (N, 3).
294    fun : Callable[[float, float, float], Any]
295        Function evaluated as `fun(x, y, z)` at each node.
296
297    Returns
298    -------
299    torch.Tensor
300		Tensor with dimension `(grid.n_nodes, Any)` with dtype `torch.float64`.
301
302    Notes
303    -----
304    This evaluates `fun` in a Python loop; it is not vectorized.
305    """
306	dummy = fun(0,0,0)  # to check the return shape
307	shape_fun = np.shape(dummy)
308	shape = (grid.n_nodes,) + shape_fun
309	field = to.zeros(shape, dtype=to.float64)
310	coordinates = grid.mesh.geometry.x
311	for i, coord in enumerate(coordinates):
312		x, y, z = coord
313		field[i] = fun(x, y, z)
314	return field

316def create_field_elems(grid: GridHandlerGMSH, fun: Fn) -> to.Tensor:
317	"""
318    Sample a N-dimensional field at element centroids using a Python callable.
319
320    Parameters
321    ----------
322    grid : GridHandlerGMSH
323        Grid handler with attributes `mesh` and `n_elems`. Assumes a tetrahedral
324        mesh where the 3→0 connectivity returns 4 vertex indices per cell.
325    fun : Callable[[float, float, float], Any]
326        Function evaluated as `fun(x, y, z)` at each cell centroid.
327
328    Returns
329    -------
330    torch.Tensor
331        Tensor with dimension `(grid.n_elems, Any)` with dtype `torch.float64`.
332
333    Notes
334    -----
335    The centroid is computed as the arithmetic mean of the 4 vertex coordinates
336    for each tetrahedral cell. This evaluates `fun` in a Python loop.
337    """
338	dummy = fun(0,0,0)  # to check the return shape
339	shape_fun = np.shape(dummy)
340	shape = (grid.n_elems,) + shape_fun
341	field = to.zeros(shape, dtype=to.float64)
342	coordinates = grid.mesh.geometry.x
343	conn_aux = grid.mesh.topology.connectivity(3, 0)
344	conn = conn_aux.array.reshape((grid.n_elems, 4))
345	for i in range(grid.n_elems):
346		cell_vertices = conn[i]
347		x = sum(coordinates[v] for v in cell_vertices) / len(cell_vertices)
348		field[i] = fun(x[0], x[1], x[2])
349	return field

352def apply_grey_theme(fig, axes, transparent=True, grid_color="0.92", back_color='0.85'):
353	fig.patch.set_facecolor("#212121ff")
354	if transparent:
355		fig.patch.set_alpha(0.0)
356	for ax in axes:
357		if ax != None:
358			ax.grid(True, color=grid_color)
359			ax.set_axisbelow(True)
360			ax.spines['bottom'].set_color('black')
361			ax.spines['top'].set_color('black')
362			ax.spines['right'].set_color('black')
363			ax.spines['left'].set_color('black')
364			ax.tick_params(axis='x', colors='black', which='both')
365			ax.tick_params(axis='y', colors='black', which='both')
366			ax.yaxis.label.set_color('black')
367			ax.xaxis.label.set_color('black')
368			ax.set_facecolor(back_color)

253	"""
254    σ = C : ε using Voigt notation (tensorial Voigt, no engineering factors).
255
256    Parameters
257    ----------
258    C_voigt : (N, 6, 6) torch.Tensor (float64)
259        Stiffness matrix per element in tensorial Voigt order [xx, yy, zz, xy, xz, yz].
260    eps_tensor : (N, 3, 3) torch.Tensor (float64)
261        Small-strain tensor per element.
262
263    Returns
264    -------
265    (N, 3, 3) torch.Tensor (float64)
266        Cauchy stress tensor per element.
267    """
268	n_elems = C_voigt.shape[0]
269	eps_voigt = to.zeros((n_elems, 6), dtype=to.float64)
270	eps_voigt[:,0] = eps_tensor[:,0,0]
271	eps_voigt[:,1] = eps_tensor[:,1,1]
272	eps_voigt[:,2] = eps_tensor[:,2,2]
273	eps_voigt[:,3] = eps_tensor[:,0,1]
274	eps_voigt[:,4] = eps_tensor[:,0,2]
275	eps_voigt[:,5] = eps_tensor[:,1,2]

278	stress_torch[:,0,0] = stress_voigt[:,0]

281	stress_torch[:,0,1] = stress_torch[:,1,0] = stress_voigt[:,3]
282	stress_torch[:,0,2] = stress_torch[:,2,0] = stress_voigt[:,4]
283	stress_torch[:,1,2] = stress_torch[:,2,1] = stress_voigt[:,5]
284	return stress_torch

def create_field_elems( grid: safeincave.GridHandlerGMSH, fun: Callable[[float, float, float], Any]) -> torch.Tensor: 
317def create_field_elems(grid: GridHandlerGMSH, fun: Fn) -> to.Tensor:
318	"""
319    Sample a N-dimensional field at element centroids using a Python callable.
320
321    Parameters
322    ----------
323    grid : GridHandlerGMSH
324        Grid handler with attributes `mesh` and `n_elems`. Assumes a tetrahedral
325        mesh where the 3→0 connectivity returns 4 vertex indices per cell.
326    fun : Callable[[float, float, float], Any]
327        Function evaluated as `fun(x, y, z)` at each cell centroid.
328
329    Returns
330    -------
331    torch.Tensor
332        Tensor with dimension `(grid.n_elems, Any)` with dtype `torch.float64`.
333
334    Notes
335    -----
336    The centroid is computed as the arithmetic mean of the 4 vertex coordinates
337    for each tetrahedral cell. This evaluates `fun` in a Python loop.
338    """
339	dummy = fun(0,0,0)  # to check the return shape
340	shape_fun = np.shape(dummy)
341	shape = (grid.n_elems,) + shape_fun
342	field = to.zeros(shape, dtype=to.float64)
343	coordinates = grid.mesh.geometry.x
344	conn_aux = grid.mesh.topology.connectivity(3, 0)
345	conn = conn_aux.array.reshape((grid.n_elems, 4))
346	for i in range(grid.n_elems):
347		cell_vertices = conn[i]
348		x = sum(coordinates[v] for v in cell_vertices) / len(cell_vertices)
349		field[i] = fun(x[0], x[1], x[2])
350	return field
