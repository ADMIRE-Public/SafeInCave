 28class GeneralBC(ABC):
 29	"""
 30    Base container for time-dependent boundary-condition data.
 31
 32    Subclasses (e.g., :class:`DirichletBC`, :class:`NeumannBC`) set the
 33    concrete fields and `type` identifier.
 34
 35    Attributes
 36    ----------
 37    boundary_name : str or None
 38        Name/label of the boundary as defined by mesh tags.
 39    type : str or None
 40        Boundary condition type identifier (e.g., ``"dirichlet"``, ``"neumann"``).
 41    values : list[float] or None
 42        Boundary values sampled at the times in ``time_values``.
 43    time_values : list[float] or None
 44        Monotonically increasing times associated with ``values``.
 45    """
 46	def __init__(self):
 47		self.boundary_name = None
 48		self.type = None
 49		self.values = None
 50		self.time_values = None

 53class DirichletBC(GeneralBC):
 54	"""
 55    Time-dependent Dirichlet (essential) boundary condition for one component.
 56
 57    Parameters
 58    ----------
 59    boundary_name : str
 60        Named boundary in the mesh tags (handled by the grid object).
 61    component : int
 62        Component index of the vector field to constrain (e.g., 0=x, 1=y, 2=z).
 63    values : list[float]
 64        Prescribed values over time (interpolated with :func:`numpy.interp`).
 65    time_values : list[float]
 66        Times corresponding to ``values`` (must be monotone).
 67
 68    Attributes
 69    ----------
 70    type : str
 71        Always ``"dirichlet"``.
 72    component : int
 73        Constrained component index.
 74    values, time_values : list[float]
 75        Stored time history for interpolation.
 76    boundary_name : str
 77        Stored boundary label.
 78    """
 79	def __init__(self, boundary_name: str, component: int, values: list, time_values: list):
 80		self.boundary_name = boundary_name
 81		self.type = "dirichlet"
 82		self.values = values
 83		self.time_values = time_values
 84		self.component = component

 86class NeumannBC(GeneralBC):
 87	"""
 88    Time-dependent Neumann (traction/pressure) boundary condition with hydrostatics.
 89
 90    The applied boundary value is modeled as
 91    ``p(t) + ρ g (H - x[i])``, where ``p(t)`` is a time-dependent pressure,
 92    ``ρ`` is density, ``g`` is gravity (default negative for downward),
 93    ``H`` is a reference elevation, and ``x[i]`` is the spatial coordinate
 94    in the chosen direction.
 95
 96    Parameters
 97    ----------
 98    boundary_name : str
 99        Named boundary in the mesh tags.
100    direction : int
101        Coordinate index used for hydrostatic variation (0=x, 1=y, 2=z).
102    density : float
103        Fluid/solid density ``ρ``.
104    ref_pos : float
105        Reference elevation ``H``.
106    values : list[float]
107        Time samples for the base pressure ``p(t)`` (before sign).
108    time_values : list[float]
109        Times corresponding to ``values`` (must be monotone).
110    g : float, default=-9.81
111        Gravitational acceleration (sign included).
112
113    Attributes
114    ----------
115    type : str
116        Always ``"neumann"``.
117    direction : int
118        Axis index for hydrostatic term.
119    density : float
120        Stored density.
121    ref_pos : float
122        Stored reference elevation.
123    gravity : float
124        Stored gravitational acceleration.
125    boundary_name, values, time_values : as given
126        Stored metadata and time history.
127    """
128	def __init__(self, boundary_name: str, direction: int, density: float, ref_pos: float, values: list, time_values: list, g=-9.81):
129		self.boundary_name = boundary_name
130		self.type = "neumann"
131		self.values = values
132		self.time_values = time_values
133		self.direction = direction
134		self.density = density
135		self.ref_pos = ref_pos
136		self.gravity = g

139class BcHandler():
140    """
141    Boundary-condition handler for a linear momentum problem.
142
143    Stores user-defined BC objects, organizes them by type, and converts them
144    into DOLFINx/UFL objects at a given time ``t`` for assembly.
145
146    Parameters
147    ----------
148    equation : LinearMomentum
149        Momentum equation object providing:
150        - ``grid`` with mesh, boundary dimension, and tag queries,
151        - ``get_uV()`` to access the vector function space (for Dirichlet DOFs),
152        - ``normal`` (outward unit normal vector for boundary integrals),
153        - ``ds`` boundary measure.
154
155    Attributes
156    ----------
157    dirichlet_boundaries : list[DirichletBC]
158        Registered Dirichlet BCs.
159    neumann_boundaries : list[NeumannBC]
160        Registered Neumann BCs.
161    dirichlet_bcs : list[dolfinx.fem.DirichletBC] (set by :meth:`update_dirichlet`)
162        DOLFINx Dirichlet BC objects for the current time.
163    neumann_bcs : list[ufl.Form] (set by :meth:`update_neumann`)
164        UFL surface terms to add to the right-hand side.
165    x : ufl.core.expr.Expr
166        Spatial coordinate vector ``x = SpatialCoordinate(mesh)`` used for hydrostatics.
167    """
168    def __init__(self):
169        self.dirichlet_boundaries = []
170        self.neumann_boundaries = []
171        
172    def set_uV(self, uV):
173        self.uV = uV
174
175    def set_boundary_dim(self, boundary_dim):
176        self.boundary_dim = boundary_dim
177
178    def set_boudary_tags(self, boundary_tags):
179        self.boundary_tags = boundary_tags
180
181    def set_dolfin_tags(self, dolfin_tags):
182        self.dolfin_tags = dolfin_tags
183
184    def set_normal(self, normal):
185        self.normal = normal
186
187    def set_ds(self, ds):
188        self.ds = ds
189
190    def set_spatial_coordinates(self, mesh):
191        self.x = ufl.SpatialCoordinate(mesh)
192
193    def reset_boundary_conditions(self) -> None:
194        """
195        Clear all registered boundary conditions.
196
197        Returns
198        -------
199        None
200        """
201        self.dirichlet_boundaries = []
202        self.neumann_boundaries = []
203
204    def add_boundary_condition(self, bc : GeneralBC) -> None:
205        """
206        Register a boundary condition instance.
207
208        Parameters
209        ----------
210        bc : GeneralBC
211            One of :class:`DirichletBC` or :class:`NeumannBC`.
212
213        Returns
214        -------
215        None
216
217        Raises
218        ------
219        Exception
220            If the boundary condition type is not supported.
221        """
222        if bc.type == "dirichlet":
223            self.dirichlet_boundaries.append(bc)
224        elif bc.type == "neumann":
225            self.neumann_boundaries.append(bc)
226        else:
227            raise Exception(f"Boundary type {bc.type} not supported.")
228
229    def update_dirichlet(self, t: float) -> None:
230        """
231        Build Dirichlet BC objects at time ``t``.
232
233        Parameters
234        ----------
235        t : float
236            Current simulation time.
237
238        Returns
239        -------
240        None
241
242        Side Effects
243        ------------
244        Populates :attr:`dirichlet_bcs` with
245        :class:`dolfinx.fem.DirichletBC` constructed by:
246        - locating DOFs on each boundary for the target component, and
247        - interpolating the prescribed value via :func:`numpy.interp`.
248        """
249        self.dirichlet_bcs = []
250        for bc in self.dirichlet_boundaries:
251            value = np.interp(t, bc.time_values, bc.values)
252            dofs = do.fem.locate_dofs_topological(
253                self.uV.sub(bc.component),
254                self.boundary_dim,
255                self.boundary_tags[bc.boundary_name]
256            )
257            self.dirichlet_bcs.append(
258                do.fem.dirichletbc(
259                    do.default_scalar_type(value),
260                    dofs,
261                    self.uV.sub(bc.component)
262                )
263            )
264
265    def update_neumann(self, t: float) -> None:
266        """
267        Build Neumann contributions (tractions/pressures) at time ``t``.
268
269        For each :class:`NeumannBC`, the boundary term is constructed as
270        ``(p(t) + ρ g (H - x[i])) * n * ds(tag)``, where ``n`` is the outward
271        unit normal, and ``ds(tag)`` is the boundary measure for the target
272        boundary.
273
274        Parameters
275        ----------
276        t : float
277            Current simulation time.
278
279        Returns
280        -------
281        None
282
283        Side Effects
284        ------------
285        Populates :attr:`neumann_bcs` with UFL surface integrals to be added
286        to the right-hand side form.
287        """
288        self.neumann_bcs = []
289        for bc in self.neumann_boundaries:
290            i = bc.direction
291            rho = bc.density
292            H = bc.ref_pos
293            p = -np.interp(t, bc.time_values, bc.values)
294            value_neumann = p + rho*bc.gravity*(H - self.x[i])
295            self.neumann_bcs.append(
296                value_neumann*self.normal*self.ds(
297                    self.dolfin_tags[self.boundary_dim][bc.boundary_name]
298                )
299            )
300
301    def update_cavern_bcs(self, cavern_handler: CavernHandler):
302        self.cavern_bcs = []
303        for cavern in cavern_handler.caverns_PT + cavern_handler.caverns_MFlux:
304            i = cavern.direction
305            rho = cavern.density
306            H = cavern.ref_pos
307            p = -cavern.P
308            load = p + rho*cavern.gravity*(H - self.x[i])
309            self.cavern_bcs.append(
310                load*self.normal*self.ds(
311                    self.dolfin_tags[self.boundary_dim][cavern.cavern_name]
312                )
313            )

def update_neumann(self, t: float) -> None: 
266    def update_neumann(self, t: float) -> None:
267        """
268        Build Neumann contributions (tractions/pressures) at time ``t``.
269
270        For each :class:`NeumannBC`, the boundary term is constructed as
271        ``(p(t) + ρ g (H - x[i])) * n * ds(tag)``, where ``n`` is the outward
272        unit normal, and ``ds(tag)`` is the boundary measure for the target
273        boundary.
274
275        Parameters
276        ----------
277        t : float
278            Current simulation time.
279
280        Returns
281        -------
282        None
283
284        Side Effects
285        ------------
286        Populates :attr:`neumann_bcs` with UFL surface integrals to be added
287        to the right-hand side form.
288        """
289        self.neumann_bcs = []
290        for bc in self.neumann_boundaries:
291            i = bc.direction
292            rho = bc.density
293            H = bc.ref_pos
294            p = -np.interp(t, bc.time_values, bc.values)
295            value_neumann = p + rho*bc.gravity*(H - self.x[i])
296            self.neumann_bcs.append(
297                value_neumann*self.normal*self.ds(
298                    self.dolfin_tags[self.boundary_dim][bc.boundary_name]
299                )
300            )
