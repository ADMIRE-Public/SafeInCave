```latex
@software{honorio_2026_20722991,
  author       = {Honório, Hermínio and
                  Hajibeygi, Hadi},
  title        = {SafeInCave v3.0.3},
  month        = jun,
  year         = 2026,
  publisher    = {Zenodo},
  version      = {v3.0.3},
  doi          = {10.5281/zenodo.20722991},
  url          = {https://doi.org/10.5281/zenodo.20722991},
}
```
