 23def build_smoother(points: np.ndarray, conn: np.ndarray) -> np.ndarray:
 24    """
 25    Build a node–cell smoothing operator for tetrahedral meshes.
 26
 27    Given node coordinates and tetrahedral connectivities, this constructs a
 28    (sparse) smoothing matrix that averages cell-centered quantities to nodes
 29    and back to cells via a volume-weighted scheme.
 30
 31    Parameters
 32    ----------
 33    points : (n_nodes, 3) ndarray of float
 34        Node coordinates (x, y, z).
 35    conn : (n_elems, 4) ndarray of int
 36        Tetrahedral connectivity (node indices per element).
 37
 38    Returns
 39    -------
 40    smoother : (n_elems, n_elems) scipy.sparse.csr_matrix
 41        Sparse smoothing operator mapping cell fields to smoothed cell fields.
 42        (Note: although the annotation says ``np.ndarray``, the returned
 43        object is a CSR sparse matrix.)
 44
 45    Notes
 46    -----
 47    The operator is assembled as ``B * A`` where:
 48    - ``A`` distributes cell values to nodes using volume weights of incident
 49      cells.
 50    - ``B`` averages nodal values back to cells with uniform weights over the
 51      element's nodes.
 52
 53    See Also
 54    --------
 55    scipy.sparse.csr_matrix
 56    """
 57    def tetrahedron_volume(x1, y1, z1, x2, y2, z2, x3, y3, z3, x4, y4, z4):
 58        """Compute signed volume of a tetrahedron given its four vertices."""
 59        volume = abs((1/6) * ((x2 - x1) * ((y3 - y1)*(z4 - z1) - (z3 - z1)*(y4 - y1)) + 
 60                     (y2 - y1) * ((z3 - z1)*(x4 - x1) - (x3 - x1)*(z4 - z1)) + 
 61                     (z2 - z1) * ((x3 - x1)*(y4 - y1) - (y3 - y1)*(x4 - x1))))
 62        return volume
 63
 64    def build_node_elem_stencil(conn, coord):
 65        """Build node-to-incident-element adjacency lists."""
 66        stencil = [[] for i in range(n_nodes)]
 67        for elem, elem_conn in enumerate(conn):
 68            for node in elem_conn:
 69                if elem not in stencil[node]:
 70                    stencil[node].append(elem)
 71        return stencil
 72
 73    # Initialize
 74    n_elems = conn.shape[0]
 75    n_nodes = points.shape[0]
 76
 77    # Calculate volumes of all tetrahedra
 78    volumes = np.zeros(n_elems)
 79    for i in range(n_elems):
 80        nodes = conn[i]
 81        x1, y1, z1 = points[nodes[0], 0], points[nodes[0], 1], points[nodes[0], 2]
 82        x2, y2, z2 = points[nodes[1], 0], points[nodes[1], 1], points[nodes[1], 2]
 83        x3, y3, z3 = points[nodes[2], 0], points[nodes[2], 1], points[nodes[2], 2]
 84        x4, y4, z4 = points[nodes[3], 0], points[nodes[3], 1], points[nodes[3], 2]
 85        volumes[i] = tetrahedron_volume(x1, y1, z1, x2, y2, z2, x3, y3, z3, x4, y4, z4)
 86
 87    stencil = build_node_elem_stencil(conn, points)
 88
 89    A_row, A_col, A_data = [], [], []
 90    for node in range(n_nodes):
 91        vol = volumes[stencil[node]].sum()
 92        for elem in stencil[node]:
 93            A_row.append(node)
 94            A_col.append(elem)
 95            A_data.append(volumes[elem]/vol)
 96    A_csr = csr_matrix((A_data, (A_row, A_col)), shape=(n_nodes, n_elems))
 97
 98    B_row, B_col, B_data = [], [], []
 99    for elem, nodes in enumerate(conn):
100        for node in nodes:
101            B_row.append(elem)
102            B_col.append(node)
103            B_data.append(1/len(nodes))
104    B_csr = csr_matrix((B_data, (B_row, B_col)), shape=(n_elems, n_nodes))
105    smoother = B_csr.dot(A_csr)
106
107    return smoother

109def build_mapping(nodes_xdmf: np.ndarray, nodes_msh: np.ndarray) -> list[int]:
110    """
111    Build an index mapping from XDMF node order to MSH node order.
112
113    For each coordinate triplet in ``nodes_xdmf``, finds the row index in
114    ``nodes_msh`` with the exact same coordinates and returns the list of
115    corresponding indices.
116
117    Parameters
118    ----------
119    nodes_xdmf : (n_nodes, 3) ndarray of float
120        Node coordinates as read from an XDMF file.
121    nodes_msh : (n_nodes, 3) ndarray of float
122        Node coordinates as read from a .msh file.
123
124    Returns
125    -------
126    mapping : list of int
127        For each row in ``nodes_xdmf``, the index of the identical row in
128        ``nodes_msh``.
129
130    Notes
131    -----
132    This uses exact floating-point equality. If the two sources differ by
133    round-off, consider a tolerance-based nearest matching instead.
134    """
135    return [np.where((nodes_msh == row).all(axis=1))[0][0] for row in nodes_xdmf]

137def find_closest_point(target_point: np.ndarray, points: np.ndarray) -> int:
138    """
139    Find the index of the closest point in a set to a target point.
140
141    Parameters
142    ----------
143    target_point : (3,) ndarray of float
144        Query point (x, y, z).
145    points : (n_points, 3) ndarray of float
146        Candidate points.
147
148    Returns
149    -------
150    idx : int
151        Index of the closest point in ``points`` (Euclidean distance).
152    """
153    x_p, y_p, z_p = target_point
154    d = np.sqrt(  (points[:,0] - x_p)**2
155                + (points[:,1] - y_p)**2
156                + (points[:,2] - z_p)**2 )
157    p_idx = d.argmin()
158    return p_idx

161def compute_cell_centroids(cells: np.ndarray, points: np.ndarray) -> np.ndarray:
162    """
163    Compute centroids of tetrahedral cells.
164
165    Parameters
166    ----------
167    cells : (n_cells, 4) ndarray of int
168        Tetrahedral connectivity (node indices per cell).
169    points : (n_nodes, 3) ndarray of float
170        Node coordinates (x, y, z).
171
172    Returns
173    -------
174    centroids : (n_cells, 3) ndarray of float
175        Centroid coordinates for each cell, computed as the arithmetic mean
176        of its four vertex coordinates.
177    """
178    n_cells = cells.shape[0]
179    centroids = np.zeros((n_cells, 3))
180    for i, cell in enumerate(cells):
181        p0 = points[cell[0]]
182        p1 = points[cell[1]]
183        p2 = points[cell[2]]
184        p3 = points[cell[3]]
185        x = (p0[0] + p1[0] + p2[0] + p3[0])/4
186        y = (p0[1] + p1[1] + p2[1] + p3[1])/4
187        z = (p0[2] + p1[2] + p2[2] + p3[2])/4
188        centroids[i,:] = np.array([x, y, z])
189    return centroids

192def read_cell_tensor(xdmf_field_path: str) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
193    """
194    Read a time series of cell-centered 3x3 tensor fields from an XDMF file.
195
196    Parameters
197    ----------
198    xdmf_field_path : str
199        Path to the XDMF file containing cell data (``cells['tetra']``).
200
201    Returns
202    -------
203    centroids : (n_cells, 3) ndarray of float
204        Centroid coordinates of the tetrahedral cells.
205    time_list : (n_steps,) ndarray of float
206        Time values for each time step.
207    tensor_field : (n_steps, n_cells, 3, 3) ndarray of float
208        Tensor values per time step and cell.
209
210    Notes
211    -----
212    The function assumes a single tensor field is present under
213    ``cell_data['tetra']`` at each time step, and reshapes it to (3, 3) per cell.
214    """
215    reader = ms.xdmf.TimeSeriesReader(xdmf_field_path)
216    points, cells = reader.read_points_cells()
217    n_cells = cells["tetra"].shape[0]
218    n_steps = reader.num_steps
219
220    centroids = compute_cell_centroids(cells["tetra"], points)
221    tensor_field = np.zeros((n_steps, n_cells, 3, 3))
222    time_list = np.zeros(n_steps)
223
224    for k in range(reader.num_steps):
225        # Read data
226        time, point_data, cell_data = reader.read_data(k)
227
228        # Add time
229        time_list[k] = time
230
231        # Add tensor
232        field_name = list(cell_data["tetra"].keys())[0]
233        tensor_field[k,:,:] = cell_data["tetra"][field_name].reshape((n_cells, 3, 3))
234
235    return centroids, time_list, tensor_field
